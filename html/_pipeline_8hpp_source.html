152 pipeLine(int sw, int sh, float angle, float np, float fp, Eigen::Vector3f cp, Eigen::Vector3f tp, float scl){
153 // Manual construction of the Perspective Projection Matrix
154 projection = Eigen::Matrix4f::Zero();
155 float tn = std::tanf(angle*0.5f);
156 float as = static_cast<float>(sw) / static_cast<float>(sh);
157
158 projection(0,0) = 1.0f / (tn*as);
159 projection(1,1) = 1.0f / tn;
160 projection(2,2) = fp/(fp-np);
161 projection(2,3) = -(fp*np)/(fp-np);
162 projection(3,2) = 1.0f;
163 projection(3,3) = 0.0f;
164
165 // Manual construction of the View Matrix (LookAt System)
166 Eigen::Vector3f worldUp = Eigen::Vector3f(0.0f,1.0f,0.0f);
167 Eigen::Vector3f cameraForward = (tp - cp).normalized();
168 Eigen::Vector3f cameraRight = cameraForward.cross( worldUp).normalized();
169 Eigen::Vector3f cameraUp = cameraRight.cross(cameraForward).normalized();
170
171 view = Eigen::Matrix4f::Identity();
172
173 view.row(0).head<3>() = cameraRight.transpose();
174 view.row(1).head<3>() = cameraUp.transpose();
175 view.row(2).head<3>() = cameraForward.transpose();
176 view(0,3) = -cp.dot(cameraRight);
177 view(1,3) = -cp.dot(cameraUp);
178 view(2,3) = -cp.dot(cameraForward);
179
180 scale = scl;
181 projView = projection*view;
182 quat = Eigen::Quaternionf::Identity();
183
184 update ();
185 }
