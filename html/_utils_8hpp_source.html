68 bool eigenVV( Eigen::Matrix3f& A, Eigen::Matrix3f& V )
69 {
70 assert(A.isApprox(A.transpose()) && "A must be symmetric");
71
72 float norm = A.norm();
73 V = Eigen::Matrix3f::Identity();
74
76 if(norm == 0) return true;
77
79 float toler = std::numeric_limits<float>::epsilon()*norm;
80
82 int maxit = 100;
83
84 float tau = 0.0f;
85 float c = 0.0f; // Cosine of the angle
86 float s = 0.0f; // Sine of the angle
87 float t = 0.0f; // Tangent of the angle
88
89 int p, q, r;
90
91 for(int x=0;x<maxit;x++)
92 {
93
94 // Hardcoded search for the maximum off-diagonal value
95 // Indices of rotated axes (p,q) and the invariant axis (r)
96 p = 0; q = 1; r = 2;
97 float max_offdiag = std::abs(A(1,0));
98
99 if( std::abs(A(2,0)) > max_offdiag )
100 {
101 max_offdiag = std::abs(A(2,0));
102 p = 0; q = 2; r = 1;
103 }
104 if(std::abs(A(2,1)) > max_offdiag)
105 {
106 max_offdiag = std::abs(A(2,1));
107 p = 1; q = 2; r = 0;
108 }
109
110 // Exit condition: Off-diagonal elements are negligible
111 if (max_offdiag < toler) return true;
112
113 // Algebraic deduction of the rotation without trigonometric functions
114 // Givens rotation (without math::sin/cos/atan)
115 tau = (A(p,p)-A(q,q)) / (2.0f*A(p,q));
116 t = std::copysign(1.0f,tau) / (std::abs(tau) + std::sqrtf((tau*tau) +1));
117 c = 1.0f / std::sqrtf((t*t) + 1.0f); // cos(θ) = 1 / √ (t^2 + 1)
118 s = t*c; // tan(θ) = sin(θ) / cos(θ)
119
120 // Temporary variables for the update
121 float pp = A(p,p) + A(p,q) * t;
122 float qq = A(q,q) - A(p,q) * t;
123
124 float pr = A(p,r)*c + s * A(q,r);
125 float qr = A(p,r)*-s + c * A(q,r);
126
128 A(p,p) = pp;
129 A(q,q) = qq;
130
131 A(p,r) = pr;
132 A(q,r) = qr;
133
134 A(r,q) = qr;
135 A(r,p) = pr;
136
137 A(p,q) = 0.0f;
138 A(q,p) = 0.0f;
139
141 for(int y = 0; y < 3; y ++)
142 {
143 float yp = V(y,p)*c + V(y,q)*s;
144 float yq = -V(y,p)*s + V(y,q)*c;
145
146 V(y,p) = yp;
147 V(y,q) = yq;
148 }
149 }
150 return false;
151 }
152}
